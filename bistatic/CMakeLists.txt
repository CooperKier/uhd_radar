cmake_minimum_required(VERSION 3.16)
project(ORCA_BISTATIC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BISTATIC_BUILD_RADIO "Build the UHD radar executables" ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_bistatic_protocol ../tests/test_bistatic_protocol.cpp)
  target_compile_options(test_bistatic_protocol PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(test_bistatic_protocol PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME bistatic_protocol COMMAND test_bistatic_protocol)
endif()
if(BISTATIC_BUILD_RADIO)
  if(DEFINED ENV{CONDA_PREFIX})
    list(PREPEND CMAKE_PREFIX_PATH "$ENV{CONDA_PREFIX}")
  endif()
  find_package(UHD 4.9 REQUIRED)
  find_package(yaml-cpp REQUIRED)
  find_package(Threads REQUIRED)
  if(TARGET yaml-cpp::yaml-cpp)
    set(BISTATIC_YAML yaml-cpp::yaml-cpp)
  elseif(TARGET yaml-cpp)
    set(BISTATIC_YAML yaml-cpp)
  else()
    set(BISTATIC_YAML ${YAML_CPP_LIBRARIES})
  endif()
  add_executable(bistatic main.cpp)
  target_compile_options(bistatic PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(bistatic PRIVATE ${UHD_INCLUDE_DIRS} ${YAML_CPP_INCLUDE_DIR} ${YAML_CPP_INCLUDE_DIRS})
  target_link_libraries(bistatic PRIVATE ${UHD_LIBRARIES} ${BISTATIC_YAML} Threads::Threads)
endif()
