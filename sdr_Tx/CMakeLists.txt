cmake_minimum_required(VERSION 3.16)
project(ORCA_TX_ONLY LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(DEFINED ENV{CONDA_PREFIX})
  list(PREPEND CMAKE_PREFIX_PATH "$ENV{CONDA_PREFIX}")
endif()
find_package(UHD 4.9 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)

# Deliberately independent of the receive build and its GoogleTest download.
add_executable(radar_tx main.cpp sdr.cpp chirp.cpp rf_settings.cpp pseudorandom_phase.cpp)
target_include_directories(radar_tx PRIVATE ${UHD_INCLUDE_DIRS} ${YAML_CPP_INCLUDE_DIR} ${YAML_CPP_INCLUDE_DIRS})
if(TARGET yaml-cpp::yaml-cpp)
  set(TX_YAML_TARGET yaml-cpp::yaml-cpp)
elseif(TARGET yaml-cpp)
  set(TX_YAML_TARGET yaml-cpp)
else()
  set(TX_YAML_TARGET ${YAML_CPP_LIBRARIES})
endif()
target_link_libraries(radar_tx PRIVATE ${UHD_LIBRARIES} ${TX_YAML_TARGET} Threads::Threads)

find_package(Boost REQUIRED)
target_include_directories(radar_tx PRIVATE ${Boost_INCLUDE_DIRS})
add_executable(gps_logger gps_logger.cpp)
target_include_directories(gps_logger PRIVATE ${Boost_INCLUDE_DIRS})
target_compile_definitions(gps_logger PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
target_link_libraries(gps_logger PRIVATE Threads::Threads)

add_executable(pseudorandom_phase_codes_to_file pseudorandom_phase_to_file.cpp pseudorandom_phase.cpp)
